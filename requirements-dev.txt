# Test-suite dependencies. The host CLI tools under host/ are stdlib
# only on purpose - they have to run from the system interpreter during
# bring-up - but pytest is not stdlib, so the suite runs from a venv:
#
#   python3 -m venv .venv
#   .venv/bin/pip install -r requirements-dev.txt
#   .venv/bin/python -m pytest tests -v
pytest>=7.0
pyserial==3.5   # tools/soak0c_portable.py: the only cross-platform serial client here

# The bench oscilloscope, for the measurements the ADC under
# investigation cannot make about itself. USBTMC needs raw USB access
# and macOS has no kernel driver for it, so pyusb talks to the
# instrument directly - and libusb-package carries prebuilt libusb
# binaries, which keeps this to a pip install rather than a system
# package and a sudo. host/scope.py imports both lazily and raises
# ScopeUnavailable, so a machine with neither still collects and runs
# the whole suite.
pyusb==1.3.1
libusb-package==1.0.30.0
